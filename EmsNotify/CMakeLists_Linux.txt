cmake_minimum_required(VERSION 3.16)

project(EmsNotify VERSION 1.0.4 LANGUAGES CXX)

set(GITHUB_OWNER "harsha-deep")
set(GITHUB_REPO  "EmsNotify")

configure_file(
    include/config.h.in
    ${CMAKE_CURRENT_BINARY_DIR}/config.h
    @ONLY
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)

qt_add_executable(EmsNotify
    src/main.cpp
    src/tray_manager.cpp
    include/tray_manager.h
    resources/app.qrc
    resources/app.rc
)

target_include_directories(EmsNotify PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_BINARY_DIR}
)

target_link_libraries(EmsNotify PRIVATE
    Qt6::Core
    Qt6::Widgets
    Qt6::Network
)
